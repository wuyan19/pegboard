```rust
// crates/pegboard-core/src/config/mod.rs

use std::collections::HashMap;
use std::net::{IpAddr, SocketAddr};
use std::path::PathBuf;

#[derive(Debug, Clone)]
pub struct Config {
    pub server: ServerConfig,
    pub identity: IdentityConfig,
    pub limits: Limits,
    pub storage: StorageConfig,
}

#[derive(Debug, Clone)]
pub struct ServerConfig {
    pub listen: SocketAddr,
    pub mode: Mode,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, serde::Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum Mode { Local, Lan }

#[derive(Debug, Clone)]
pub enum IdentityConfig {
    Fixed(Option<String>),
    Forwarded { trusted: Vec<IpAddr> },
    Tokens(HashMap<String, String>),
}

#[derive(Debug, Clone)]
pub struct Limits {
    pub kv_value_bytes: u64,
    pub kv_total_bytes: u64,
    pub file_bytes: u64,
    pub file_total_bytes: u64,
    pub net_rps: u32,
    pub sign_ttl_max: u64,
}

impl Default for Limits {
    fn default() -> Self {
        Self {
            kv_value_bytes:    1 * 1024 * 1024,        // 1 MiB
            kv_total_bytes:   10 * 1024 * 1024,        // 10 MiB
            file_bytes:      100 * 1024 * 1024,        // 100 MiB
            file_total_bytes:  1 * 1024 * 1024 * 1024, // 1 GiB
            net_rps: 20,
            sign_ttl_max: 86_400,                       // 24h
        }
    }
}

#[derive(Debug, Clone)]
pub struct StorageConfig {
    pub data_root: PathBuf,
    pub apps_dir: PathBuf,
}
```

```rust
    #[serde(default)]

    #[serde(default)] pub files: bool,
    #[serde(default)] pub net: Vec<String>,
    #[serde(default)] pub ws: Vec<String>,

}

impl Default for Permissions {
    fn default() -> Self {
        Self { store: false, files: false, net: vec![], ws: vec![], shim: true }
    }
}

fn default_true() -> bool { true }

#[derive(Debug, Clone, Default, serde::Deserialize)]
pub struct LimitsOverride {
    pub store_bytes: Option<u64>,
    pub file_bytes: Option<u64>,
    pub net_rps: Option<u32>,
}

#[derive(Debug, Clone)]
pub struct AppMeta {
    pub id: String,
    pub name: String,
    pub root: PathBuf,
    pub entry: PathBuf,
    pub manifest: Manifest,
    pub limits: Limits,
    pub enabled: bool,
}
```

`merge_limits` 实现：

```rust
fn merge_limits(default: &Limits, ov: &LimitsOverride, id: &str) -> Result<Limits, AppError> {
    let mut l = default.clone();
    if let Some(v) = ov.store_bytes {
        if v > default.kv_total_bytes {
            return Err(AppError::LimitsExceeded(id.into()));
        }
        l.kv_total_bytes = v;
    }
    if let Some(v) = ov.file_bytes {
        if v > default.file_total_bytes {
            return Err(AppError::LimitsExceeded(id.into()));
        }
        l.file_total_bytes = v;
    }
    if let Some(v) = ov.net_rps {
        if v > default.net_rps {
            return Err(AppError::LimitsExceeded(id.into()));
        }
        l.net_rps = v;
    }
    Ok(l)
}
```

## 5. pegboard-server / ingress 的 `ApiError`

```rust
// crates/pegboard-server/src/ingress/error.rs

use axum::response::{IntoResponse, Response};
use axum::http::StatusCode;
use axum::Json;

#[derive(Debug)]
pub struct ApiError {
    pub code: &'static str,
    pub message: String,
    pub detail: serde_json::Value,
    pub status: StatusCode,
}

impl ApiError {
    pub fn new(code: &'static str, status: StatusCode, message: impl Into<String>) -> Self {
        Self { code, message: message.into(), detail: serde_json::Value::Null, status }
    }

    pub fn with_detail(mut self, detail: serde_json::Value) -> Self {
        self.detail = detail;
        self
    }
}

impl IntoResponse for ApiError {
    fn into_response(self) -> Response {
        let body = serde_json::json!({
            "error": {
                "code": self.code,
                "message": self.message,
                "detail": self.detail,
            }
        });
        (self.status, Json(body)).into_response()
    }
}
```

错误映射（节选）：

```rust
use pegboard_core::guard::GuardError;
use pegboard_core::store::StoreError;
use pegboard_core::files::FilesError;
use pegboard_core::proxy::ProxyError;
use pegboard_core::identity::IdentityError;

impl From<GuardError> for ApiError {
    fn from(e: GuardError) -> Self {
        match e {
            GuardError::PermissionDenied(_) =>
                ApiError::new("PERMISSION_DENIED", StatusCode::FORBIDDEN, e.to_string()),
            GuardError::TargetDenied(_) | GuardError::SsrfBlocked(_) =>
                ApiError::new("TARGET_DENIED", StatusCode::FORBIDDEN, e.to_string()),
            GuardError::RateLimited { .. } =>
                ApiError::new("LIMIT_EXCEEDED", StatusCode::TOO_MANY_REQUESTS, e.to_string()),
            GuardError::InvalidTarget(_) =>
                ApiError::new("INVALID_REQUEST", StatusCode::BAD_REQUEST, e.to_string()),
        }
    }
}

impl From<StoreError> for ApiError {
    fn from(e: StoreError) -> Self {
        match e {
            StoreError::QuotaExceeded { .. } =>
                ApiError::new("LIMIT_EXCEEDED", StatusCode::TOO_MANY_REQUESTS, e.to_string()),
            StoreError::ValueTooLarge { .. } | StoreError::KeyTooLong { .. } =>
                ApiError::new("LIMIT_EXCEEDED", StatusCode::PAYLOAD_TOO_LARGE, e.to_string()),
            StoreError::InvalidValue(_) =>
                ApiError::new("INVALID_REQUEST", StatusCode::BAD_REQUEST, e.to_string()),
            _ =>
                ApiError::new("INVALID_REQUEST", StatusCode::INTERNAL_SERVER_ERROR, e.to_string()),
        }
    }
}

impl From<FilesError> for ApiError {
    fn from(e: FilesError) -> Self {
        match e {
            FilesError::NotFound(_) =>
                ApiError::new("NOT_FOUND", StatusCode::NOT_FOUND, e.to_string()),
            FilesError::TooLarge { .. } =>
                ApiError::new("LIMIT_EXCEEDED", StatusCode::PAYLOAD_TOO_LARGE, e.to_string()),
            FilesError::QuotaExceeded { .. } =>
                ApiError::new("LIMIT_EXCEEDED", StatusCode::TOO_MANY_REQUESTS, e.to_string()),
            FilesError::TokenInvalid =>
                ApiError::new("TOKEN_INVALID", StatusCode::FORBIDDEN, e.to_string()),
            FilesError::InvalidName(_) =>
                ApiError::new("INVALID_REQUEST", StatusCode::BAD_REQUEST, e.to_string()),
            _ =>
                ApiError::new("INVALID_REQUEST", StatusCode::INTERNAL_SERVER_ERROR, e.to_string()),
        }
    }
}

impl From<ProxyError> for ApiError {
    fn from(e: ProxyError) -> Self {
        match e {
            ProxyError::Denied(g) => g.into(),
            ProxyError::Timeout =>
                ApiError::new("TIMEOUT", StatusCode::GATEWAY_TIMEOUT, e.to_string()),
            ProxyError::Upstream(_) | ProxyError::WsUpgrade(_) =>
                ApiError::new("UPSTREAM_ERROR", StatusCode::BAD_GATEWAY, e.to_string()),
            ProxyError::RedirectDenied(_) =>
                ApiError::new("TARGET_DENIED", StatusCode::FORBIDDEN, e.to_string()),
            ProxyError::TooLarge { .. } =>
                ApiError::new("LIMIT_EXCEEDED", StatusCode::PAYLOAD_TOO_LARGE, e.to_string()),
            _ =>
                ApiError::new("INVALID_REQUEST", StatusCode::BAD_REQUEST, e.to_string()),
        }
    }
}

impl From<IdentityError> for ApiError {
    fn from(e: IdentityError) -> Self {
        ApiError::new("PERMISSION_DENIED", StatusCode::UNAUTHORIZED, e.to_string())
    }
}
```

`assets/sdk.js` 骨架（可运行最小版）：

```js
(function () {
  "use strict";

  const APP_ID = (location.pathname.match(/^\/apps\/([a-z0-9_-]+)\//) || [])[1] || null;
  const NATIVE_FETCH = window.fetch.bind(window);

  class HostError extends Error {
    constructor(code, message, detail) {
      super(message);
      this.code = code;
      this.detail = detail ?? null;
    }
  }

  async function api(path, init = {}) {
    const headers = new Headers(init.headers || {});
    if (APP_ID) headers.set("X-Pegboard-App", APP_ID);
    const r = await NATIVE_FETCH(path, { ...init, headers });
    if (!r.ok) {
      let body = null;
      try { body = await r.json(); } catch (_) {}
      const e = body && body.error;
      throw new HostError(e?.code || "INVALID_REQUEST",
                          e?.message || r.statusText, e?.detail);
    }
    return r;
  }

  const store = {
    async get(k)     { const r = await api(`/api/store/kv/${encodeURIComponent(k)}`);
                       return (await r.json()).value; },
    async set(k, v)  { await api(`/api/store/kv/${encodeURIComponent(k)}`,
                       { method: "PUT", headers: { "Content-Type": "application/json" },
                         body: JSON.stringify({ value: v }) }); },
    async delete(k)  { await api(`/api/store/kv/${encodeURIComponent(k)}`, { method: "DELETE" }); },
    async list(p)    { const r = await api(`/api/store/kv?prefix=${encodeURIComponent(p || "")}`);
                       return (await r.json()).items; },
    async batch(ops) { await api(`/api/store/kv/batch`,
                       { method: "POST", headers: { "Content-Type": "application/json" },
                         body: JSON.stringify({ ops }) }); },
  };

  const files = {
    async upload(file) {
      const fd = new FormData();
      fd.append("file", file, file.name || "blob");
      const r = await api("/api/files", { method: "POST", body: fd });
      return r.json();
    },
    async get(id)  { const r = await api(`/api/files/${id}`);
                     return r.ok ? r.blob() : null; },
    url(id)        { return `/api/files/${id}`; },
    async sign(id, ttl) {
      const r = await api(`/api/files/${id}/sign`,
        { method: "POST", headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ ttl }) });
      return (await r.json()).url;
    },
    async list(opts = {}) {
      const q = new URLSearchParams();
      if (opts.prefix) q.set("prefix", opts.prefix);
      if (opts.cursor) q.set("cursor", opts.cursor);
      if (opts.limit)  q.set("limit", opts.limit);
      const r = await api(`/api/files?${q}`);
      return r.json();
    },
    async delete(id) { await api(`/api/files/${id}`, { method: "DELETE" }); },
  };

  async function hostFetch(url, options) {
    const r = await api(`/api/proxy?url=${encodeURIComponent(url)}`, options);
    return r;
  }

  function connectWS(url) {
    const proto = location.protocol === "https:" ? "wss:" : "ws:";
    const q = encodeURIComponent(url);
    return new WebSocket(`${proto}//${location.host}/api/ws-proxy?url=${q}`);
  }

  function hostUrl(url) {
    return `/api/asset?url=${encodeURIComponent(url)}`;
  }

  window.host = { store, files, fetch: hostFetch, connectWS, url: hostUrl, user: undefined };

  if (window.__pegboard_shim !== false) {
    window.fetch = function (input, init) {
      const url = typeof input === "string" ? input : input.url;
      try {
        const u = new URL(url, location.href);
        if (u.origin === location.origin) return NATIVE_FETCH(input, init);
        return hostFetch(url, init);
      } catch (_) {
        return NATIVE_FETCH(input, init);
      }
    };
  }
})();
```
